Funcionamiento de la Encriptación

El proceso secuencial paso a paso explicado de forma clara y directa para cualquier audiencia.

¿Cómo Opera en Python?

En Python, el proceso funciona tomando el mensaje original y pasándolo a través de una librería de seguridad junto con una clave secreta, lo que transforma el texto normal en un resultado revuelto e ilegible.

Esa librería se encarga de hacer todo el trabajo matemático por dentro para que no tengas que inventar nada complicado. Después, cuando alguien con permiso quiere leer el mensaje, vuelve a usar la misma clave para hacer el camino inverso y recuperar el texto original tal como estaba al principio.

from cryptography.fernet import Fernet

# 1. Generar la llave secreta
llave = Fernet.generate_key()
caja = Fernet(llave)

# 2. Cifrar el texto original (en bytes)
secreto_oculto = caja.encrypt(b"Mi clave del WiFi es 1234")
print("Asi se ve encriptado:", secreto_oculto)

# 3. Descifrar y recuperar el original
secreto_abierto = caja.decrypt(secreto_oculto)
print("Asi vuelve a verse:", secreto_abierto.decode())

import base64
import os
from getpass import getpass

from cryptography.exceptions import InvalidTag
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.hazmat.primitives.kdf.argon2 import Argon2id


def derivar_clave(contrasena: bytes, sal: bytes) -> bytes:
    kdf = Argon2id(
        salt=sal,
        length=32,
        iterations=3,
        lanes=4,
        memory_cost=64 * 1024,
    )
    return kdf.derive(contrasena)

def cifrar(mensaje: str, contrasena: bytes) -> str:
    sal = os.urandom(16)
    nonce = os.urandom(12)
    clave = derivar_clave(contrasena, sal)

    texto_bytes = mensaje.encode()
    cifrado = AESGCM(clave).encrypt(nonce, texto_bytes, None)
    paquete = sal + nonce + cifrado
    return base64.urlsafe_b64encode(paquete).decode("ascii")


def descifrar(token: str, contrasena: bytes) -> str:
    paquete = base64.urlsafe_b64decode(token.encode("ascii"))
    sal = paquete[:16]
    nonce = paquete[16:28]
    cifrado = paquete[28:]

    clave = derivar_clave(contrasena, sal)
    texto_bytes = AESGCM(clave).decrypt(nonce, cifrado, None)
    return texto_bytes.decode()

def main():
    tokens = {}
    siguiente_id = 1

    while True:
        print("\n=== Demo Argon2id + AES-GCM ===")
        print("1. Cifrar un mensaje")
        print("2. Descifrar un mensaje de esta sesión")
        print("3. Borrar mensajes cifrados")
        print("4. Salir")
        opcion = input("Elige 1, 2, 3 o 4: ").strip()


        if opcion == "1":
            contrasena = getpass("Contraseña: ").encode()
            if not contrasena:
                print("La contraseña no puede estar vacía.")
                continue

            mensaje = input("Mensaje para cifrar: ")
            token = cifrar(mensaje, contrasena)
            tokens[siguiente_id] = token

            print(f"\nToken de cifrado #{siguiente_id}:")
            print(token)
            siguiente_id += 1

        elif opcion == "2":
            if not tokens:
                print("Primero cifra un mensaje con la opción 1.")
                continue

            print("Tokens disponibles:", ", ".join(map(str, tokens.keys())))
            try:
                token_id = int(input("Número del token para descifrar: "))
                token = tokens[token_id]
            except (ValueError, KeyError):
                print("Ese token no está en el historial.")
                continue

            contrasena = getpass("Contraseña usada para cifrarlo: ").encode()
            try:
                mensaje = descifrar(token, contrasena)
            except InvalidTag:
                print("Contraseña incorrecta o token alterado.")
            else:
                print("\nToken recuperado")
                print(f"Token de cifrado #{token_id}:")
                print(token)
                print("Mensaje descifrado:", mensaje)

        elif opcion == "3":
            if input("¿Borrar historial? (s/n): ").strip().lower() == "s":
                tokens.clear()
                print("Historial borrado.")

        elif opcion == "4":
            print("Demo terminada.")
            break

        else:
            print("Opción inválida.")

if __name__ == "__main__":
    main()

Algoritmo ganador de la Password Hashing Competition. Requiere 64 MB de memoria y 4 lanes, neutralizando ataques masivos paralelos con GPU y ASIC.

Modo Galois/Counter Mode (AEAD). Cifra el mensaje con clave de 256 bits y genera un tag MAC que detecta alteraciones lanzando InvalidTag.

# Cifrado César didáctico
def cifrado_cesar(texto, desplazamiento=3):
    resultado = []
    for char in texto:
        if char.isalpha():
            base = ord('A') if char.isupper() else ord('a')
            nuevo = chr((ord(char) - base + desplazamiento) % 26 + base)
            resultado.append(nuevo)
        else:
            resultado.append(char)
    return "".join(resultado)

mensaje = "Mi clave del WiFi es 1234"
print("Cifrado:", cifrado_cesar(mensaje, desplazamiento=3))

Conclusiones

Ideas centrales y reflexiones sobre la encriptación y su aplicación con Python.

Esencial en el Mundo Digital

La encriptación es esencial para proteger la información en el mundo digital. Con algoritmos como AES y RSA, y librerías como Cryptography y PyCryptodome, Python permite cifrar contraseñas, archivos y mensajes sin implementar matemáticas complejas desde cero.

Pilar: Algoritmos Estándar & Productividad

Confidencialidad e Integridad

El cifrado garantiza la confidencialidad, ya que solo quien tiene la clave correcta puede recuperar el mensaje, y la integridad, al detectar modificaciones no autorizadas. Esto es clave en la banca en línea, la mensajería privada y el almacenamiento de datos personales.

Pilar: Privacidad + No Manipulación

Gestión de Claves y Buenas Prácticas

Sin embargo, la seguridad depende también de la gestión de claves: una clave expuesta compromete todo el sistema. Por eso conviene usar librerías reconocidas en lugar de crear algoritmos propios.
